   ```java
   private List<ContainerStatusTargetResultItem> convert2TargetResultItem(List<ForceOptRoomStatusDO> roomStatusDOList) {
       if (CollectionUtils.isEmpty(roomStatusDOList)) {
           return Lists.newArrayList();
       }
       List<ContainerStatusTargetResultItem> resultItemList = Lists.newArrayListWithCapacity(roomStatusDOList.size());
       for (ForceOptRoomStatusDO roomStatusDO : roomStatusDOList) {
           ContainerStatusTargetResultItem resultItem = new ContainerStatusTargetResultItem();
           resultItem.setPoiId(roomStatusDO.getPoiId());
           resultItem.setPartnerId(roomStatusDO.getPartnerId());
           resultItem.setContainerId(roomStatusDO.getContainerId());
           resultItem.setCheckinDate(roomStatusDO.getCheckinDate());
           resultItem.setContainerStatus(ContainerStatusEnum.valid);
   
           if (roomStatusDO.getOptType().equals(OptRoomStatusEnum.FORCE_OPT_OPEN_ROOM.getCode()) || roomStatusDO.getOptType().equals(OptRoomStatusEnum.FORCE_OPT_CLOSE_ROOM.getCode())) {
               resultItem.setOrderType(OrderTypeEnum.FORCE_OPT_ORDER);
           }
   
           if (roomStatusDO.getOptType().equals(OptRoomStatusEnum.TRACE_OPEN_ROOM.getCode()) || roomStatusDO.getOptType().equals(OptRoomStatusEnum.TRACE_CLOSE_ROOM.getCode())) {
               resultItem.setOrderType(OrderTypeEnum.TRACE_ROOM_ORDER);
           }
   
           resultItemList.add(resultItem);
       }
       return resultItemList;
   }
   ```

4. 组装数据 container_id，check_in_date，room_id（mta_room_id），写入  pegasus / hotel_pegasus.force_opt_room_status 表

   组装数据

   ```JAVA
   private ForceOptRoomStatusDO convert2ForceOptRoomStatusDO(Date now, ForceOptRoomDTO roomDTO) {
       ForceOptRoomStatusDO roomStatusDO = new ForceOptRoomStatusDO();
       roomStatusDO.setGmtModify(now);
       if (Objects.equals(roomDTO.getForceOptRoomEnum(),ForceOptRoomEnum.FORCE_OPEN_SUC)
               || Objects.equals(roomDTO.getForceOptRoomEnum(),ForceOptRoomEnum.ROOM_REPEAT)
               || Objects.equals(roomDTO.getForceOptRoomEnum(),ForceOptRoomEnum.TRACE_ROOM_SUC)){
           roomStatusDO.setStatus(CONTAINER_VALID);
       }else {
           roomStatusDO.setStatus(CONTAINER_INVALID);
       }
       roomStatusDO.setContainerId(roomDTO.getContainerId());
       roomStatusDO.setPoiId(roomDTO.getPoiId());
       roomStatusDO.setPartnerId(roomDTO.getPartnerId());
       roomStatusDO.setRoomId(roomDTO.getRoomId());
       roomStatusDO.setCheckinDate(roomDTO.getCheckinDate());
       roomStatusDO.setDatekey(DateUtils.date2Int(now, DateUtils.YYYYMMDD));
       roomStatusDO.setContainerCheckinDate(Long.valueOf("" + roomDTO.getContainerId() + roomDTO.getCheckinDate()));
       roomStatusDO.setGmtCreate(now);
       roomStatusDO.setUploadTime(roomDTO.getUploadTime());
       roomStatusDO.setOptRes(roomDTO.getForceOptRoomEnum().getCode());
       OptRoomBizType optRoomType = OptRoomContext.getValue().getOptRoomType();
       if (optRoomType == null) {
           optRoomType = OptRoomBizType.FORCE_OPT_ROOM;
       }
       roomStatusDO.setOptType(optRoomType.getCode());
       roomStatusDO.setStrategyId(roomDTO.getStrategyId());
       roomStatusDO.setStrategyName(roomDTO.getStrategyName());
       roomStatusDO.setConditionId(roomDTO.getConditionId());
       roomStatusDO.setChosenType(roomDTO.getChosenType());
       roomStatusDO.setGoodsType(null == roomDTO.getGoodsType() ? GoodsTypeEnum.dayRoom.getValue() : roomDTO.getGoodsType());
       return roomStatusDO;
   }
   ```
